# Capstone Project

* initialize npm
  npm init -y

# Backend set up
  * Installed packages
  npm install -D typescript @^6.0.0
  npm install -D tsx@4.23.0 
  npm install -D @types/node
  npm i express
  npm i -D @types/express
  npm i dotenv
  npm install pino
  npm install -D pino-pretty
  npm i cors
  npm i -D @types/cors

* initialized tsconfig.json
  npx tsc --init

* verify tailwind version
  npx tsc -v

* entry file
  app.ts
